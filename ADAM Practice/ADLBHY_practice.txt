    SET ADAM.ADSL_V1;

DATA lb_sdtm;
	SET LBSDTM.lb;

DATA adlbhy_std;
	SET ADLBHY.adlbhy;
RUN;

DATA ADLBC_V1;
	SET ADAM.adlbc_v1;
RUN;

PROC SQL;
SELECT
    STRIP(name)||'='||QUOTE(TRIM(label))
INTO :LABEL_ADLBHY SEPARATED BY ' '
FROM Dictionary.columns
WHERE libname = 'WORK'
AND memname = 'ADLBHY_STD'
;
QUIT;

%PUT &LABEL_ADLBHY;

PROC SQL;
SELECT name
INTO :ADLBHY_NAME SEPARATED BY ' '
FROM Dictionary.columns
WHERE libname = 'WORK'
AND memname = 'ADLBHY_STD'
;
QUIT;

%PUT &ADLBHY_NAME;


/*===================================================================================================
						Derivation Chain 0
==================================================================================================*/

/*Step 1 Derive variables from ADSL directly*/
DATA ADSL_STEP_1;
	LENGTH RACE $32;
	SET ADSL_V1;
	TRTP = TRT01P;
	TRTPN = TRT01PN;
	TRTA = TRT01A;
	TRTAN = TRT01AN;
	KEEP STUDYID SUBJID USUBJID TRTP TRTPN TRTA TRTAN TRTSDT TRTEDT 
	AGE AGEGR1 AGEGR1N RACE RACEN SEX SAFFL COMP24FL DSRAEFL SAFFL;
RUN;

/*Step 2 Derive variables from ADLBC*/

DATA ADLBC_FOR_HYLAW;
	SET ADLBC_V1;
	WHERE PARAMCD IN ('ALT','AST','BILI') 
	AND AVISITN IN (0,2,4,6,8,12,16,20,24);
RUN;

PROC FREQ DATA = ADLBHY_STD;
	TABLES AVISIT*AVISITN / LIST MISSING;
RUN;

PROC FREQ DATA = ADLBC_FOR_HYLAW;
	TABLES AVISIT*AVISITN / LIST MISSING;
RUN;

/*Step 3 Derive PARAM_LOOKUP*/

DATA param_lookup;
    LENGTH PARAMTYP $7 PARAM $100 PARAMCD $8 PARCAT1 $5;
    INFILE DATALINES DLM='|' DSD;
    INPUT PARAMTYP $ PARAM $ PARAMCD $ PARAMN PARCAT1 $;
DATALINES;
 |Alanine Aminotransferase (U/L)|ALT|1|CHEM
 |Aspartate Aminotransferase (U/L)|AST|2|CHEM
 |Bilirubin (umol/L)|BILI|3|CHEM
DERIVED|Bilirubin 1.5 x ULN|BILIHY|4|HYLAW
DERIVED|Transaminase 1.5 x ULN|TRANSHY|5|HYLAW
DERIVED|Total Bili 1.5 x ULN and Transaminase 1.5 x ULN|HYLAW|6|HYLAW
;
RUN;

PROC PRINT DATA = param_lookup;
RUN;


/*=====================================================================================
							Derive variables from the ADLBC without HYLAW
=======================================================================================*/
/*Step 4 Derive SHIFT1 SHIFT1N CRIT1 CRIT1FL CRIT1FN*/

DATA ADLBC_STEP_4;
	SET ADLBC_FOR_HYLAW;
	LENGTH SHIFT1 $16;
	LENGTH SHIFT1N 8;
	LENGTH CRIT1 $12;
	LENGTH CRIT1FL $1;
	LENGTH CRIT1FN 8;
	SHIFT1 = '';
	SHIFT1N = .;
	CRIT1 = 'R2A1HI > 1.5';
	IF R2A1HI > 1.5 THEN CRIT1FL = 'Y';
		ELSE IF NOT MISSING(R2A1HI) AND R2A1HI <= 1.5 THEN CRIT1FL = 'N';
		ELSE CRIT1FL = '';
	IF CRIT1FL = 'Y' THEN CRIT1FN = 1;
	ELSE IF CRIT1FL = 'N' THEN CRIT1FN = 0;
	ELSE CRIT1FN = .;
RUN;


/*=========================================================================================
							Derive variables from the ADLBC with HYLAW
============================================================================================*/
/*BILIHY*/
/*Step 5 Derive AVAL, PARAMCD = BILIHY */
DATA BILIHY_STEP_5;
	SET ADLBC_STEP_4;
	OUTPUT;
	
	IF PARAMCD = 'BILI' AND R2A1HI > 1.5 THEN DO;
		AVAL = 1;
		PARAMCD = 'BILIHY';
	OUTPUT;
	END;
	ELSE IF PARAMCD = 'BILI' AND R2A1HI <= 1.5 AND NOT MISSING(R2A1HI) THEN DO;
		AVAL = 0;
		PARAMCD = 'BILIHY';
	OUTPUT;
	END;
	ELSE IF PARAMCD = 'BILI' AND MISSING(R2A1HI) THEN DO;
		AVAL = .;
		PARAMCD = 'BILIHY';
	OUTPUT; 
	END;
RUN;

PROC FREQ DATA = BILIHY_STEP_5;
	TABLES PARAMCD / MISSING;
RUN;

/*Step 6 Derive BASE, A1LO, A1HI, R2A1LO, R2A1HI, BR2A1LO, BR2A1HI for PARAMCD = BILIHY */
PROC SORT DATA = BILIHY_STEP_5;
	BY STUDYID USUBJID PARAMCD AVISITN;
RUN;


DATA BILIHY_STEP_6;
	SET BILIHY_STEP_5;
	BY STUDYID USUBJID PARAMCD AVISITN;
	RETAIN BASE_VAL;
	
	IF FIRST.PARAMCD = 1 THEN BASE_VAL = .;
	IF ABLFL = 'Y' THEN BASE_VAL = AVAL;
	
	IF PARAMCD = 'BILIHY' THEN DO;
		IF NOT MISSING(AVAL) THEN BASE = BASE_VAL;
		ELSE CALL MISSING(BASE);
		CALL MISSING(A1LO, A1HI, R2A1LO, R2A1HI, BR2A1LO, BR2A1HI, ADY, ADT, VISIT, VISITNUM);
	END;
	DROP CHG BASE_VAL;
RUN;

/*Step 7 Derive AVAL for PARAMCD = TRANSHY */
/*Transpose for data */
DATA TRANSHY_FOR_TRANS;
	SET BILIHY_STEP_6;
	WHERE PARAMCD IN ('ALT','AST');
RUN;

PROC SORT DATA = TRANSHY_FOR_TRANS;
	BY USUBJID PARAMCD AVISITN;
RUN;

PROC TRANSPOSE DATA = TRANSHY_FOR_TRANS OUT = TRANS_COMPARE (DROP = _NAME_ _LABEL_);
	BY USUBJID PARAMCD AVISITN;
	VAR R2A1HI;
	ID PARAMCD;
RUN;

/*Filter R2A1HI > 1.5 in ALT*/

DATA ALT_FL;
	SET TRANS_COMPARE;
	WHERE PARAMCD = 'ALT';
	IF ALT > 1.5 THEN ALT_FL = 'Y';
	ELSE IF ALT <= 1.5 AND NOT MISSING(ALT) THEN ALT_FL = 'N';
	ELSE ALT_FL = '';
	DROP AST;
RUN;

DATA ADLBC_FOR_MERGE_ABLFL_ALT;
	SET adlbc_for_hylaw;
	WHERE PARAMCD IN ('ALT');
	KEEP USUBJID PARAMCD AVISITN ABLFL;
RUN;

PROC SORT DATA = ALT_FL;
	BY USUBJID AVISITN;
RUN;

PROC SORT DATA = ADLBC_FOR_MERGE_ABLFL_ALT;
	BY USUBJID AVISITN;
RUN;

DATA ALT_ABLFL_FINAL;
	MERGE ALT_FL(IN = A) ADLBC_FOR_MERGE_ABLFL_ALT(IN = B);
	BY USUBJID AVISITN;
	IF A = 1;
	IF B = 0 THEN PUT 'WARNING: ABLFL 未配對成功 ' USUBJID= AVISITN=;
RUN;


/*Filter R2A1HI > 1.5 in AST*/

DATA AST_FL;
	SET TRANS_COMPARE;
	WHERE PARAMCD = 'AST';
	IF AST > 1.5 THEN AST_FL = 'Y';
	ELSE IF AST <= 1.5 AND NOT MISSING(AST) THEN AST_FL = 'N';
	ELSE AST_FL = '';
	DROP ALT;
RUN;

DATA ADLBC_FOR_MERGE_ABLFL_AST;
	SET adlbc_for_hylaw;
	WHERE PARAMCD IN ('AST');
	KEEP USUBJID PARAMCD AVISITN ABLFL;
RUN;

PROC SORT DATA = AST_FL;
	BY USUBJID AVISITN;
RUN;

PROC SORT DATA = ADLBC_FOR_MERGE_ABLFL_AST;
	BY USUBJID AVISITN;
RUN;

DATA AST_ABLFL_FINAL;
	MERGE AST_FL(IN = A) ADLBC_FOR_MERGE_ABLFL_AST(IN = B);
	BY USUBJID AVISITN;
	IF A = 1;
	IF B = 0 THEN PUT 'WARNING: ABLFL 未配對成功 ' USUBJID= AVISITN=;
RUN;



/*Merge AST_ABLFL_FINAL and ALT_ABLFL_FINAL*/

PROC SORT DATA = AST_ABLFL_FINAL;
	BY USUBJID AVISITN;
RUN;

PROC SORT DATA = ALT_ABLFL_FINAL;
	BY USUBJID AVISITN;
RUN;

DATA TRANS_ABLFL;
	MERGE ALT_ABLFL_FINAL (IN = A) AST_ABLFL_FINAL(IN = B);
	BY USUBJID AVISITN;
	IF A = 1;
	
	IF ALT_FL = 'Y' OR AST_FL = 'Y' THEN AVAL = 1;
	ELSE IF MISSING(ALT_FL) OR MISSING(AST_FL) THEN AVAL = .;
	ELSE AVAL = 0;
	DROP PARAMCD ALT AST ALT_FL AST_FL ;
RUN;



/*Filter the dataset from ADSL*/
PROC SORT DATA = TRANS_ABLFL;
	BY USUBJID;
RUN;

PROC SORT DATA = ADSL_STEP_1;
	BY USUBJID;
RUN;

DATA TRANSHY_MERGE_ADSL;
	MERGE TRANS_ABLFL(IN = A) ADSL_STEP_1(IN = B);
	BY USUBJID;
	IF A = 1;
	PARAMCD = 'TRANSHY';
RUN;

/*Stack the datasets*/

DATA TRANSHY_STEP_7;
	SET BILIHY_STEP_6 TRANSHY_MERGE_ADSL;
RUN;


/*Step 8 Derive BASE, A1LO, A1HI, R2A1LO, R2A1HI, BR2A1LO, BR2A1HI for PARAMCD = TRANSHY */
PROC SORT DATA = TRANSHY_STEP_7;
	BY USUBJID PARAMCD AVISITN;
RUN;

DATA TRANSHY_STEP_8;
	SET TRANSHY_STEP_7;
	RETAIN BASE_VAL;
	BY USUBJID PARAMCD AVISITN;
	
	IF FIRST.PARAMCD = 1 THEN BASE_VAL = .;
	
	IF ABLFL = 'Y' THEN BASE_VAL = AVAL;
	
	IF PARAMCD = 'TRANSHY' THEN DO;
		CALL MISSING(A1LO, A1HI, R2A1HI, R2A1LO, BR2A1HI, BR2A1LO);
		BASE = BASE_VAL;
	END;
	DROP BASE_VAL;
RUN;


/*Step 9 Derive AVAL for PARAMCD = HYLAW */

DATA HYLAW_FOR_TRANS;
	SET TRANSHY_STEP_8;
	WHERE PARAMCD IN ('BILIHY', 'TRANSHY');
RUN;

/*Transpose for data */

PROC SORT DATA = HYLAW_FOR_TRANS;
	BY USUBJID PARAMCD AVISITN;
RUN;

PROC TRANSPOSE DATA = HYLAW_FOR_TRANS OUT = HYLAW_COMPARE (DROP = _NAME_ _LABEL_);
	BY USUBJID PARAMCD AVISITN;
	VAR AVAL;
	ID PARAMCD;
RUN;

/*Filter BILIHY AVAL = 1 and ABLFL*/

DATA BILIHY_FL;
	SET HYLAW_COMPARE;
	WHERE PARAMCD = 'BILIHY';
	IF BILIHY = 1 THEN BILI_FL = 'Y';
	ELSE IF MISSING(BILIHY) THEN BILI_FL = '';
	ELSE IF BILIHY = 0 THEN BILI_FL = 'N';
	DROP TRANSHY;
RUN;

DATA ADLBC_FOR_MERGE_ABLFL_BILI;
	SET adlbc_for_hylaw;
	WHERE PARAMCD IN ('BILI');
	KEEP USUBJID PARAMCD AVISITN ABLFL;
RUN;

PROC SORT DATA = BILIHY_FL;
	BY USUBJID AVISITN;
RUN;

PROC SORT DATA = ADLBC_FOR_MERGE_ABLFL_BILI;
	BY USUBJID AVISITN;
RUN;

DATA BILIHY_ABLFL_FINAL;
	MERGE BILIHY_FL(IN = A) ADLBC_FOR_MERGE_ABLFL_BILI(IN = B);
	BY USUBJID AVISITN;
	IF A = 1;
	IF B = 0 THEN PUT 'WARNING: ABLFL 未配對成功 ' USUBJID= AVISITN=;
RUN;

/*Filter TRANSHY AVAL = 1 and ABLFL*/
DATA TRANSHY_FL;
	SET HYLAW_COMPARE;
	WHERE PARAMCD = 'TRANSHY';
	IF TRANSHY = 1 THEN TRANSHY_FL = 'Y';
	ELSE IF MISSING(TRANSHY) THEN TRANSHY_FL = '';
	ELSE IF TRANSHY = 0 THEN TRANSHY_FL = 'N';
	DROP BILIHY;
RUN;


DATA ADLBC_FOR_MERGE_ABLFL_TRANSHY;
	SET HYLAW_FOR_TRANS;
	WHERE PARAMCD IN ('TRANSHY');
	KEEP USUBJID PARAMCD AVISITN ABLFL;
RUN;

PROC SORT DATA = TRANSHY_FL;
	BY USUBJID AVISITN;
RUN;

PROC SORT DATA = ADLBC_FOR_MERGE_ABLFL_TRANSHY;
	BY USUBJID AVISITN;
RUN;

DATA TRANSHY_ABLFL_FINAL;
	MERGE TRANSHY_FL(IN = A) ADLBC_FOR_MERGE_ABLFL_TRANSHY(IN = B);
	BY USUBJID AVISITN;
	IF A = 1;
	IF B = 0 THEN PUT 'WARNING: ABLFL 未配對成功 ' USUBJID= AVISITN=;
RUN;

/*Merge the two datasets*/

PROC SORT DATA = TRANSHY_ABLFL_FINAL;
	BY USUBJID AVISITN;
RUN;

PROC SORT DATA = BILIHY_ABLFL_FINAL;
	BY USUBJID AVISITN;
RUN;

DATA HYLAW_ABLFL;
	MERGE TRANSHY_ABLFL_FINAL(IN = A) BILIHY_ABLFL_FINAL(IN = B);
	BY USUBJID AVISITN;
	IF A = 1;
	
	IF TRANSHY_FL = 'N' OR BILI_FL = 'N' THEN AVAL = 0;
	ELSE IF TRANSHY_FL = 'Y' AND BILI_FL = 'Y' THEN AVAL = 1;
	ELSE AVAL = .;

	DROP PARAMCD TRANSHY BILIHY TRANSHY_FL BILI_FL;
RUN;

/*Filter the dataset from ADSL*/
PROC SORT DATA = HYLAW_ABLFL;
	BY USUBJID;
RUN;

PROC SORT DATA = ADSL_STEP_1;
	BY USUBJID;
RUN;

DATA HYLAW_MERGE_ADSL;
	MERGE HYLAW_ABLFL(IN = A) ADSL_STEP_1(IN = B);
	BY USUBJID;
	IF A = 1;
	PARAMCD = 'HYLAW';
RUN;

/*Stack the datasets*/

DATA HYLAW_STEP_9;
	SET TRANSHY_STEP_8 HYLAW_MERGE_ADSL;
RUN;

/*Step 10 Derive BASE, A1LO, A1HI, R2A1LO, R2A1HI, BR2A1LO, BR2A1HI for PARAMCD = HYLAW */

PROC SORT DATA = HYLAW_STEP_9;
	BY USUBJID PARAMCD AVISITN;
RUN;

DATA HYLAW_STEP_10;
	SET HYLAW_STEP_9;
	RETAIN BASE_VAL;
	BY USUBJID PARAMCD AVISITN;
	
	IF FIRST.PARAMCD = 1 THEN BASE_VAL = .;
	
	IF ABLFL = 'Y' THEN BASE_VAL = AVAL;
	
	IF PARAMCD = 'HYLAW' THEN DO;
		CALL MISSING(A1LO, A1HI, R2A1HI, R2A1LO, BR2A1HI, BR2A1LO);
		BASE = BASE_VAL;
	END;
	DROP BASE_VAL;
RUN;

/*STEP 11 Merge dataset and param_lookup*/

/*STEP 11a:param_lookup 合併(這段被你刪掉了,要補回來)*/
PROC SORT DATA = PARAM_LOOKUP; BY PARAMCD; RUN;
PROC SORT DATA = HYLAW_STEP_10; BY PARAMCD; RUN;

DATA HYLAW_STEP_11;
	MERGE PARAM_LOOKUP(IN = A) HYLAW_STEP_10(IN = B DROP = PARAM PARAMN PARCAT1);
	BY PARAMCD;
	IF B = 1;
RUN;

/*STEP 11b:AVISIT 對照表合併*/
PROC SORT DATA = ADLBC_FOR_HYLAW(KEEP = AVISITN AVISIT) OUT = AVISIT_LOOKUP NODUPKEY;
	BY AVISITN;
RUN;

PROC SORT DATA = HYLAW_STEP_11 OUT = HYLAW_STEP_11S; BY AVISITN; RUN;

DATA HYLAW_STEP_11B;
	MERGE HYLAW_STEP_11S(IN = B DROP = AVISIT) AVISIT_LOOKUP;
	BY AVISITN;
	IF B;
RUN;

/*STEP 12 Derive SHIFT1, SHIFT1N, CRIT1, CRIT1FL*/
DATA STEP_12_HYLAW;
	SET HYLAW_STEP_11B;
	WHERE PARAMCD IN ('BILIHY','TRANSHY','HYLAW');
	
	IF BASE = 1 AND AVAL = 0 THEN SHIFT1 = 'High to Normal';
	ELSE IF BASE = 0 AND AVAL = 0 THEN SHIFT1 = 'Normal to Normal';
	ELSE IF BASE = 0 AND AVAL = 1 THEN SHIFT1 = 'Normal to High';
	
	IF SHIFT1 = 'High to Normal' THEN SHIFT1N = 0;
	ELSE IF SHIFT1 = 'Normal to Normal' THEN SHIFT1N = 1;
	ELSE IF SHIFT1 = 'Normal to High' THEN SHIFT1N = 2;
	
	CRIT1 = '';
	CRIT1FL = '';
	CRIT1FN = .;
RUN;

DATA STEP_12_ADLBH;
	SET HYLAW_STEP_11B;
	WHERE PARAMCD IN ('AST','ALT','BILI');
RUN;

/*STEP 13 Derive SHIFT1, SHIFT1N, CRIT1, CRIT1FL*/
DATA STEP_13_FINAL;
	RETAIN &ADLBHY_NAME;
	SET STEP_12_HYLAW STEP_12_ADLBH;
	
	LABEL &LABEL_ADLBHY;
	
	KEEP &ADLBHY_NAME;
RUN;

PROC CONTENTS DATA = STEP_13_FINAL;
RUN;

PROC CONTENTS DATA = ADLBHY_STD;
RUN;

PROC SORT DATA = ADLBHY_STD;
	BY USUBJID PARAMCD AVISITN;
RUN;

PROC SORT DATA = STEP_13_FINAL;
	BY USUBJID PARAMCD AVISITN;
RUN;

PROC COMPARE BASE = ADLBHY_STD COMPARE = STEP_13_FINAL;
	ID USUBJID PARAMCD AVISITN;
RUN;




/*STEP 14 QC*/

	SET STEP_13_FINAL;
RUN;
