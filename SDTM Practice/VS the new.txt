/*Step 1 Remove IT prefix*/

PROC SQL;
SELECT 
TRIM(name)||"="||TRIM(SUBSTR(name,4)) as removeprefix
INTO :removeprefix SEPARATED BY " "
FROM Dictionary.columns
WHERE libname = "WORK"
AND memname = "VS_RAW"
AND SUBSTR(name, 1, 3) = "IT_"
;
QUIT;

%PUT &removeprefix;

DATA vs_col_remove_IT;
	SET vs_raw;
	RENAME &removeprefix;
RUN;

PROC PRINT DATA = vs_col_remove_IT (OBS=20);
RUN;

/*Step 9 Create new table vs_conbine*/

PROC SQL;
SELECT 
name, length
FROM Dictionary.columns
WHERE libname = "WORK"
AND memname IN("VS_HEIGHT" ,"VS_WEIGHT", "VS_TEMP" ,"VS_SYSBP" ,"VS_DIABP" ,"VS_PULSE")
AND name IN ("VSTEST", "VSORRESU")
;
QUIT;


DATA vs_combine;
	RETAIN STUDYID USUBJID DOMAIN INSTANCE 
           VSTESTCD VSTEST VSORRES VSORRESU 
           VSPOS VSLOC VSDTC VSTPT;
	LENGTH VSTEST $24.;
	LENGTH VSORRESU $9.;
	SET vs_HEIGHT vs_WEIGHT vs_TEMP vs_SYSBP vs_DIABP vs_PULSE;
	VSDTC = PUT(INPUT(VSDTC,DATE11.),YYMMDD10.);
RUN;


/*Step 10 Check outliers*/

DATA vs_outliers_check;
	SET vs_combine;
	VSORRES_num = INPUT(VSORRES, best.);
	IF VSTESTCD = "WEIGHT" AND (VSORRES_num < 70 OR VSORRES_num > 350) THEN OUTLIER_CHECK = "Y";
	ELSE IF VSTESTCD = "HEIGHT" AND (VSORRES_num < 50 OR VSORRES_num > 80) THEN OUTLIER_CHECK = "Y";
	ELSE IF VSTESTCD = "TEMP" AND (VSORRES_num < 90 OR VSORRES_num > 105) THEN OUTLIER_CHECK = "Y";
	;
RUN;

PROC PRINT DATA = vs_outliers_check;
	WHERE OUTLIER_CHECK = "Y";
RUN;

DATA vs_unit_change;
	SET vs_outliers_check;
	WHERE OUTLIER_CHECK = "Y";
	LENGTH VSORRES_METRIC 8;
    IF VSTESTCD = "WEIGHT" THEN VSORRES_METRIC = VSORRES_num*2.20462;   
    ELSE IF VSTESTCD = "HEIGHT" THEN VSORRES_METRIC = VSORRES_num / 2.54; 
    ELSE IF VSTESTCD = "TEMP" THEN VSORRES_METRIC = VSORRES_num* 9/5 + 32;
RUN;

PROC PRINT DATA = vs_unit_change;

RUN;

/* 先建立一個對照表,列出哪些記錄要改成什麼值 */
DATA correction_list;
    INPUT USUBJID $ VSTESTCD $ VSDTC :$10. NEW_VALUE;
    DATALINES;
701-1015 HEIGHT 2013-01-06 148
701-1023 WEIGHT 2014-07-29 55.5

/* 合併回主資料集,依照對照表更新對應記錄 */
PROC SORT DATA = vs_combine; 
	BY USUBJID VSTESTCD VSDTC;
RUN;
PROC SORT DATA = correction_list;
	BY USUBJID VSTESTCD VSDTC;
RUN;

DATA vs_combine_updated;
    MERGE vs_combine correction_list(IN=in_correction);
    BY USUBJID VSTESTCD VSDTC;
    IF in_correction THEN VSORRES = PUT(NEW_VALUE, BEST.);
RUN;
