/*Step 1 Remove IT prefix*/


PROC PRINT DATA = ds_raw;
RUN;

PROC SQL;
SELECT
TRIM(name)||"="||TRIM(SUBSTR(name,4)) as removeprefix
INTO :removeprefix SEPARATED BY " "
FROM Dictionary.columns
WHERE libname = "WORK"
AND memname = "DS_RAW"
AND SUBSTR(name,1,3) = 'IT_'
;
QUIT;

%PUT &removeprefix;

DATA ds_removed_prefix;
	SET ds_raw;
	RENAME &removeprefix;
RUN;

PROC PRINT DATA = ds_removed_prefix;
RUN;

/*Step 2 Rename column name to standard and drop unnecessary columns*/

DATA ds_rename_col;
	SET ds_removed_prefix;
	DROP SITENM FORM FORML;
	RENAME 
	STUDY = STUDYID
	PATNUM = USUBJID
	DSSTD = DSSTDTC
	INSTANCE = VISIT;
RUN;


/*Step 3 Rename column name by aCRF*/
DATA ds_rename_CRF;
    SET ds_rename_col;
    LENGTH DSDTC $19. DSSTDTC $10.;
    IF MISSING(OTHERSP) THEN DO;
        DSDECOD = DSDEC;
        DSTERM = DSTER;
        IF DSDEC = "Randomized" THEN DSCAT = "PROTOCOL MILESTONE";
        ELSE DSCAT = "DISPOSITION EVENT";
    END;
    
    ELSE DO;
        DSDECOD = OTHERSP;
        DSTERM = OTHERSP;
        DSCAT = "OTHER EVENT";
    END;
    
    DSSTDTC = PUT(INPUT(DSSTDTC,MMDDYY10.),YYMMDD10.);
    
    IF MISSING(DSTMCOL) THEN 
    DSDTC = PUT(INPUT(DSDTCOL,MMDDYY10.),YYMMDD10.);
	ELSE 
    DSDTC = TRIM(PUT(INPUT(DSDTCOL,MMDDYY10.),YYMMDD10.))||"T"||TRIM(DSTMCOL);
RUN;

PROC PRINT DATA = ds_rename_CRF;
RUN;

/*Step 4 DROP columns unnecessary columns and UPCASE*/
DATA ds_drop_col;
	SET ds_rename_CRF;
	DROP DSTER DSDEC OTHERSP DSDTCOL DSDTCOL DSTMCOL DEATHDT;
	VISIT = UPCASE(VISIT);
	DSDECOD = UPCASE(DSDECOD);
	DSTERM = UPCASE(DSTERM);
	DOMAIN = "DS";
RUN;
